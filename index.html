from dataclasses import dataclass
from typing import Dict, List
from google import genai

MODEL_ID = "gemini-3.8-flash"

@dataclass
class AgentSpec:
    key: str
    name: str
    role_prompt: str

AGENTS: List[AgentSpec] = [
    AgentSpec(
        key="planner",
        name="Planner Agent",
        role_prompt="You are the Lead Planner Agent. Break the user objective into 3-4 concrete milestones, required components, and clear acceptance criteria."
    ),
    AgentSpec(
        key="researcher",
        name="Researcher Agent",
        role_prompt="You are the Technical Researcher Agent. Given the objective and plan, outline the optimal architecture pattern, edge cases, and complexity trade-offs."
    ),
    AgentSpec(
        key="builder",
        name="Builder Agent",
        role_prompt="You are the Senior Builder Agent. Produce clean, runnable code and configuration that satisfies the Planner's milestones and Researcher's constraints."
    ),
    AgentSpec(
        key="reviewer",
        name="Reviewer Agent",
        role_prompt="You are the QA Critic & Security Reviewer Agent. Audit the Builder's implementation against the Planner's criteria, list fixes, and assign a quality score out of 100."
    ),
]

class MultiAgentOrchestrator:
    def __init__(self, api_key: str | None = None):
        self.client = genai.Client(api_key=api_key) if api_key else genai.Client()

    def run_pipeline(self, objective: str) -> Dict[str, str]:
        context_accumulator = f"USER OBJECTIVE:\n{objective}\n"
        outputs: Dict[str, str] = {}

        for spec in AGENTS:
            interaction = self.client.interactions.create(
                model=MODEL_ID,
                system_instruction=spec.role_prompt,
                input=context_accumulator,
            )
            step_text = interaction.output_text or ""
            outputs[spec.key] = step_text
            context_accumulator += f"\n\n--- OUTPUT FROM {spec.name.upper()} ---\n{step_text}"

        return outputs
Option 1 • Fastest Live Link vercel.json + api/orchestrate.js

Deploy to Vercel (Frontend + Serverless API)

  1. Extract agentic-multiagent-studio.zip and push the folder to a new GitHub repository.
  2. Go to vercel.com/new and import your GitHub repository.
  3. Leave build settings as default (Vercel automatically detects index.html and api/orchestrate.js).
  4. Optional: Add GEMINI_API_KEY in Vercel Environment Variables to enable server-side live Gemini calls, or use the built-in Simulated Engine with zero keys!
  5. Click Deploy — you immediately get a shareable https://your-project.vercel.app live link.
Option 2 • Python FastAPI Web Service render.yaml + server.py

Deploy to Render (Python FastAPI + Static UI)

  1. Push the repository to GitHub (includes server.py, agents.py, requirements.txt, and render.yaml).
  2. On dashboard.render.com, click New → Web Service and connect your GitHub repo.
  3. Set Build Command to pip install -r requirements.txt and Start Command to uvicorn server:app --host 0.0.0.0 --port $PORT.
  4. Add GEMINI_API_KEY under Environment Variables (optional; falls back cleanly to Simulated Engine).
  5. Click Create Web Service to publish your live .onrender.com URL.
